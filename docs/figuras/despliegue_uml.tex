% ── Macro: nodo 3D UML ───────────────────────────────────────
% \umlnode{ancho}{alto}{x}{y}{Nombre del Nodo}
\newcommand{\umlnode}[5]{%
  \pgfmathsetmacro{\ox}{0.30}
  \pgfmathsetmacro{\oy}{0.30}
  % Cara superior
  \fill[white, draw=black, line width=0.8pt]
    (#3,      #4+#2)
    -- (#3+\ox, #4+#2+\oy)
    -- (#3+#1+\ox, #4+#2+\oy)
    -- (#3+#1, #4+#2) -- cycle;
  % Cara lateral derecha
  \fill[gray!20, draw=black, line width=0.8pt]
    (#3+#1, #4)
    -- (#3+#1+\ox, #4+\oy)
    -- (#3+#1+\ox, #4+#2+\oy)
    -- (#3+#1, #4+#2) -- cycle;
  % Cara frontal
  \draw[draw=black, line width=0.8pt, fill=white]
    (#3, #4) rectangle (#3+#1, #4+#2);
  % Nombre subrayado (dentro, parte superior)
  \node[font=\small\sffamily, anchor=north west] at (#3+0.15, #4+#2-0.10)
    {\underline{#5}};
}

% ── Macro: componente UML ─────────────────────────────────────
% \umlcomp{x}{y}{ancho}{alto}{Texto}
\newcommand{\umlcomp}[5]{%
  % Caja
  \draw[draw=black, line width=0.7pt, fill=white]
    (#1, #2) rectangle (#1+#3, #2+#4);
  % Pin superior
  \draw[draw=black, line width=0.6pt, fill=white]
    (#1-0.24, #2+#4*0.60) rectangle (#1+0.06, #2+#4*0.82);
  % Pin inferior
  \draw[draw=black, line width=0.6pt, fill=white]
    (#1-0.24, #2+#4*0.25) rectangle (#1+0.06, #2+#4*0.47);
  % Texto
  \node[font=\footnotesize\sffamily, align=center,
        text width=(#3-0.3)*1cm]
    at (#1+#3/2, #2+#4/2) {#5};
}
